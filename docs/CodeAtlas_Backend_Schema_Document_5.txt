4. Primary Keys & Foreign Keys
Users(user_id PK)

Organizations(org_id PK)

Repositories(repo_id PK)
organization_id → Organizations.org_id

AnalysisJobs(job_id PK)
repo_id → Repositories.repo_id

Capabilities(capability_id PK)
job_id → AnalysisJobs.job_id

UserJourneys(journey_id PK)
job_id → AnalysisJobs.job_id

LogicGaps(gap_id PK)
job_id → AnalysisJobs.job_id

Reports(report_id PK)
job_id → AnalysisJobs.job_id
